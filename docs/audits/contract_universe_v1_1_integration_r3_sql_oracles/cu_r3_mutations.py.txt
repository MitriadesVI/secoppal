 ('M1_exact_to_like','app/core/soql_builder.py',"equalities = [f\"{field} = '{cls._escape(name)}'\" for name in names]","equalities = [f\"unaccent(UPPER({field})) LIKE '{cls._escape(name)}'\" for name in names]",['C09','C18','C19','C23']),

]
results=[]
for name,path,old,new,cs in mutations:
 root=base/name;root.mkdir()
 with tarfile.open(archive) as tf:tf.extractall(root,filter='data')
 for file in changed:
  if file.startswith('tests/'):shutil.copyfile(W/file,root/file)
 p=root/path;s=p.read_text();assert s.count(old)==1,(name,s.count(old));p.write_text(s.replace(old,new))
 selected=[n for n in ids if clusters[n] in cs]
 if name=='M4_fingerprint_collision':selected+=['tests/test_contract_universe_r3_sql_oracles.py::test_contract_universe_changes_fingerprint_and_version_is_semantic']

 print('KILL',name,killed,flush=True)
 shutil.rmtree(root)
(E/'MUTATIONS.json').write_text(json.dumps(results,indent=2)+'\n')

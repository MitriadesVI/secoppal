ENV = {**os.environ, 'BURR_TELEMETRY_ENABLED': 'false'}


def run(name, cmd, cwd=W):
    t = time.monotonic()
    r = subprocess.run(cmd, cwd=cwd, env=ENV, capture_output=True, text=True)
    (E / (name + '.json')).write_text(json.dumps(dict(
        command=cmd, cwd=str(cwd), exit_code=r.returncode,
        duration_seconds=time.monotonic() - t, stdout=r.stdout, stderr=r.stderr,
    ), indent=2) + '\n')
    (E / (name + '.txt')).write_text(r.stdout + r.stderr)
    summary = (r.stdout + r.stderr).strip().splitlines()[-1] if (r.stdout + r.stderr).strip() else ''
    print(name, r.returncode, '|', summary[:120], flush=True)
    return r


def pytest(name, ids, extra=()):
    return run(name, [P, '-m', 'pytest', '-p', 'tests.contract_universe_sql_dates',
                      *ids, '-q', '--tb=short', *extra,
                      '--junitxml=' + str(E / (name + '.xml'))])


def ids_from(r4_name):
    cmd = json.load(open(R4 / (r4_name + '.json')))['command']
    return [a for a in cmd if a.startswith('tests/')]
